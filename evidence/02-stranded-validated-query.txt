-- Source: db/init/001_schema.sql (https://github.com/sanjay-complyance/pse-support-lab/blob/main/db/init/001_schema.sql)
$ docker compose exec postgres psql -U invoicely -d invoicely -c "SELECT s.name AS seller_name, i.status, COUNT(*) AS invoice_count, MIN(i.created_at) AS first_seen, MAX(i.created_at) AS last_seen FROM invoices i JOIN sellers s ON s.id = i.seller_id WHERE i.status = 'Validated' GROUP BY s.name, i.status ORDER BY invoice_count DESC;"
